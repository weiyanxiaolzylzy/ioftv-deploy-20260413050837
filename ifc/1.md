def _focus_on_element(self, idx):
    self.focus_mode = True           # 开启聚焦模式
    self.focused_idx = idx           # 记录聚焦的构件索引
    
    # 计算构件中心
    verts = self.vertices_list[idx]
    vertices = verts.reshape(-1, 3)
    self.center = vertices.mean(axis=0).tolist()  # 构件质心
    
    # 调整缩放比例（让构件占满视图）
    elem_size = np.max(vertices.max(axis=0) - vertices.min(axis=0))
    self.scale = 1.5 / max(elem_size, 1.0)
    
    self.selected_indices = {idx}
    self.update()
